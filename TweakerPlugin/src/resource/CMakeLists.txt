# Embedded assets: generate a Windows .rc from TweakerPlugin/assets, then link
# that .rc into each PE (DLL / smoke). The static lib only holds the loader.

set(TWEAKER_ASSETS_DIR "${CMAKE_SOURCE_DIR}/assets")
set(TWEAKER_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
set(TWEAKER_ASSETS_RC "${TWEAKER_GENERATED_DIR}/assets.rc")

file(MAKE_DIRECTORY "${TWEAKER_GENERATED_DIR}")

# .jpg/.hdr are here for the sake of skybox art, which is the only asset class where the source
# format is the artist's choice rather than ours - stb_image decodes all three (see
# skybox/sky_cubemap.cxx, which tone maps .hdr itself). Note that .exr is deliberately absent: stb
# has no EXR decoder at all, so embedding one would only produce a resource nothing can read.
#
# Embedding stays for small bundled art. High-resolution skies belong on disk via skybox_file= - a
# 2048px-per-face panorama is an 8192x4096 image, and nobody wants that inside the DLL.
file(GLOB_RECURSE TWEAKER_ASSET_FILES CONFIGURE_DEPENDS
    "${TWEAKER_ASSETS_DIR}/*.png"
    "${TWEAKER_ASSETS_DIR}/*.jpg"
    "${TWEAKER_ASSETS_DIR}/*.hdr"
    "${TWEAKER_ASSETS_DIR}/*.svg"
    "${TWEAKER_ASSETS_DIR}/*.ttf"
    "${TWEAKER_ASSETS_DIR}/*.txt"

    # .hlsli goes in as text, not as a compiled shader - it is a header, and it is embedded so that
    # a user's own .hlsl on disk can `#include "sky_common.hlsli"` and get the same constant layout
    # the bundled programs use, without a copy of the file next to it. sky_compile resolves includes
    # against these first (see its include_handler).
    "${TWEAKER_ASSETS_DIR}/*.hlsli"
)

set(TWEAKER_RC_CONTENT "// Auto-generated by src/resource/CMakeLists.txt — do not edit.\n")
foreach(ABS_PATH IN LISTS TWEAKER_ASSET_FILES)
    file(RELATIVE_PATH REL_PATH "${TWEAKER_ASSETS_DIR}" "${ABS_PATH}")
    string(REPLACE "\\" "/" REL_PATH "${REL_PATH}")

    # Typo folder guard (historical duplicate of fonts/).
    if(REL_PATH MATCHES "^fonst/")
        continue()
    endif()

    get_filename_component(EXT "${ABS_PATH}" EXT)
    string(TOLOWER "${EXT}" EXT_LOWER)
    # Use TW_* type names — bare FONT/TEXT collide with RC language keywords.
    if(EXT_LOWER STREQUAL ".png" OR EXT_LOWER STREQUAL ".jpg" OR EXT_LOWER STREQUAL ".hdr")
        set(RC_TYPE "TW_TEXTURE")
    elseif(EXT_LOWER STREQUAL ".svg")
        set(RC_TYPE "TW_SVG")
    elseif(EXT_LOWER STREQUAL ".ttf")
        set(RC_TYPE "TW_FONT")
    elseif(EXT_LOWER STREQUAL ".txt" OR EXT_LOWER STREQUAL ".hlsli")
        set(RC_TYPE "TW_TEXT")
    else()
        continue()
    endif()

    string(REPLACE "\\" "/" ABS_FWD "${ABS_PATH}")
    string(APPEND TWEAKER_RC_CONTENT "\"${REL_PATH}\" ${RC_TYPE} \"${ABS_FWD}\"\n")
endforeach()

# Shaders are the one asset class we compile rather than copy: assets/shaders/*.hlsl goes through
# fxc at build time and only the bytecode is embedded (as TW_SHADER).
#
# Build time rather than in the game process on purpose. Bytecode always works; runtime HLSL
# compilation needs d3dcompiler_47.dll, and making the skies we ship ourselves depend on a DLL that
# may not be there would trade a guarantee for nothing. Runtime compilation belongs to user-authored
# shaders, which is a later phase - see Docs/Internal/skybox-procedural.md.
#
# The profile comes from the filename: *.vs.hlsl -> vs_3_0, *.ps.hlsl -> ps_3_0. Shader Model 3.0 is
# the ceiling Audiosurf's device can offer, and ps_3_0 may not be paired with anything but vs_3_0 -
# which is why the vertex half exists at all.
set(_fxc_hints "")
foreach(_sdk_root "$ENV{WindowsSdkVerBinPath}" "$ENV{WindowsSdkDir}")
    if(_sdk_root)
        list(APPEND _fxc_hints "${_sdk_root}/x86" "${_sdk_root}/x64")
    endif()
endforeach()

# Fallback for a plain (non-developer-prompt) configure, where neither variable is set.
file(GLOB _fxc_sdk_globs
    "C:/Program Files (x86)/Windows Kits/10/bin/*/x64"
    "C:/Program Files (x86)/Windows Kits/10/bin/*/x86"
    "C:/Program Files/Windows Kits/10/bin/*/x64"
    "C:/Program Files/Windows Kits/10/bin/*/x86"
)
list(APPEND _fxc_hints ${_fxc_sdk_globs})

find_program(TWEAKER_FXC NAMES fxc HINTS ${_fxc_hints})

file(GLOB_RECURSE TWEAKER_SHADER_SOURCES CONFIGURE_DEPENDS "${TWEAKER_ASSETS_DIR}/shaders/*.hlsl")

# Shared headers are .hlsli so this glob does not pick them up as programs. They cannot be tracked
# through the source file the way #includes normally are - CMake does not scan HLSL for them - so
# every program depends on all of them. Coarse, but there are two of these files and recompiling a
# 30-instruction shader costs nothing.
file(GLOB_RECURSE TWEAKER_SHADER_HEADERS CONFIGURE_DEPENDS "${TWEAKER_ASSETS_DIR}/shaders/*.hlsli")

if(TWEAKER_SHADER_SOURCES AND NOT TWEAKER_FXC)
    message(FATAL_ERROR
        "fxc.exe not found, but assets/shaders/ contains HLSL to compile.\n"
        "It ships with the Windows SDK (Windows Kits/10/bin/<version>/x86). Either install the SDK "
        "component, configure from a Visual Studio developer prompt, or pass -DTWEAKER_FXC=<path>.")
endif()

set(TWEAKER_SHADER_OUTPUTS "")
foreach(SHADER_SRC IN LISTS TWEAKER_SHADER_SOURCES)
    file(RELATIVE_PATH REL_PATH "${TWEAKER_ASSETS_DIR}" "${SHADER_SRC}")
    string(REPLACE "\\" "/" REL_PATH "${REL_PATH}")

    if(REL_PATH MATCHES "\\.vs\\.hlsl$")
        set(SHADER_PROFILE vs_3_0)
    elseif(REL_PATH MATCHES "\\.ps\\.hlsl$")
        set(SHADER_PROFILE ps_3_0)
    else()
        message(FATAL_ERROR "${REL_PATH}: shader sources must be named *.vs.hlsl or *.ps.hlsl — "
                            "the profile is taken from the name.")
    endif()

    string(REGEX REPLACE "\\.hlsl$" ".fxo" REL_OUT "${REL_PATH}")
    set(SHADER_OUT "${TWEAKER_GENERATED_DIR}/${REL_OUT}")
    get_filename_component(SHADER_OUT_DIR "${SHADER_OUT}" DIRECTORY)
    file(MAKE_DIRECTORY "${SHADER_OUT_DIR}")

    add_custom_command(
        OUTPUT "${SHADER_OUT}"
        COMMAND "${TWEAKER_FXC}" /nologo /T ${SHADER_PROFILE} /E main /O3
                /I "${TWEAKER_ASSETS_DIR}/shaders" /Fo "${SHADER_OUT}" "${SHADER_SRC}"
        DEPENDS "${SHADER_SRC}" ${TWEAKER_SHADER_HEADERS}
        COMMENT "fxc ${REL_PATH} -> ${SHADER_PROFILE}"
        VERBATIM
    )

    list(APPEND TWEAKER_SHADER_OUTPUTS "${SHADER_OUT}")
    string(APPEND TWEAKER_RC_CONTENT "\"${REL_OUT}\" TW_SHADER \"${SHADER_OUT}\"\n")

    # The source goes in beside the bytecode, as text. Not for compiling - that already happened -
    # but because a shader's parameter annotations live in its own source (see skybox/sky_params),
    # and a built-in program has to be able to read its own block exactly as a user's .hlsl does.
    string(APPEND TWEAKER_RC_CONTENT "\"${REL_PATH}\" TW_TEXT \"${SHADER_SRC}\"\n")
endforeach()

# The .rc is written to a staging file at configure time and copied to its real path by a build
# rule that depends on every asset it names.
#
# That indirection is not ceremony. rc.exe reads the files the .rc points at, but emits no
# dependency information, so nothing tells the build system that recompiling is needed when one of
# them changes - and the .rc's own text does not change, because it holds paths rather than content.
# The result is a resource blob that silently keeps whatever it was built with: an edited shader,
# icon or font simply does not reach the DLL. This cost an afternoon of debugging a shader that was
# correct on disk and stale in the binary.
#
# Copying (rather than copy_if_different) is deliberate - the point is to move the timestamp, which
# is the only thing that makes the .res recompile.
set(TWEAKER_ASSETS_RC_IN "${TWEAKER_ASSETS_RC}.in")
file(WRITE "${TWEAKER_ASSETS_RC_IN}" "${TWEAKER_RC_CONTENT}")

add_custom_command(
    OUTPUT "${TWEAKER_ASSETS_RC}"
    COMMAND "${CMAKE_COMMAND}" -E copy "${TWEAKER_ASSETS_RC_IN}" "${TWEAKER_ASSETS_RC}"
    DEPENDS "${TWEAKER_ASSETS_RC_IN}" ${TWEAKER_ASSET_FILES} ${TWEAKER_SHADER_SOURCES} ${TWEAKER_SHADER_OUTPUTS}
    COMMENT "Refreshing embedded assets"
    VERBATIM
)

add_library(tweaker_resource STATIC
    resource.cxx
    self_extract.cxx
)

target_include_directories(tweaker_resource PUBLIC
    "${CMAKE_SOURCE_DIR}/src"
)

# Same pch.hxx as the DLL (needs DX/Detours/Quest3D include paths for the PCH, plus libuulog since
# pch.hxx pulls in uulog.hh).
target_link_libraries(tweaker_resource PRIVATE
    dxsdk
    detours
    quest3d_sdk
    libuulog
    luajit_headers
)
target_precompile_headers(tweaker_resource PRIVATE "${CMAKE_SOURCE_DIR}/src/pch.hxx")

# luajit_headers only carries include paths, and one of those headers (luajit.h) is generated during
# the luajit target's build. The PCH compile has to wait for it, which an INTERFACE library cannot
# express on its own.
add_dependencies(tweaker_resource luajit)

# Expose the generated .rc path so smoke (and others) can embed it in their PE.
set(TWEAKER_ASSETS_RC "${TWEAKER_ASSETS_RC}" CACHE INTERNAL "Path to generated TweakerPlugin assets.rc")

target_link_libraries(TweakerPlugin PRIVATE tweaker_resource)
target_sources(TweakerPlugin PRIVATE "${TWEAKER_ASSETS_RC}")

# The .rc names the .fxo by path, so they have to exist before rc runs. The custom target carries
# that ordering to other PEs embedding the same .rc (smoke); the rebuild itself rides on the copy
# rule above, whose output is the .rc the compiler is actually given.
add_custom_target(tweaker_shaders DEPENDS ${TWEAKER_SHADER_OUTPUTS} "${TWEAKER_ASSETS_RC}")
add_dependencies(TweakerPlugin tweaker_shaders)
